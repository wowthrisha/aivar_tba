# FEATURE A (intelligence-v6) — isolated venv for app/mcp_server.py.
#
# Deliberately NOT merged into the project's root requirements.txt: the
# `mcp` package (both its v2 line and the legacy v1.x line, both tried)
# pulls in a starlette major version incompatible with this project's
# pinned fastapi==0.115.0 (confirmed by direct reproduction - installing
# mcp[cli] into the same environment as app.main breaks
# `from app.main import app` with `TypeError: Router.__init__() got an
# unexpected keyword argument 'on_startup'`). app/mcp_server.py works
# around this by never importing app.main - it's a thin HTTP client over
# the same REST API any other MCP-compatible agent would use - so it can
# run in its own venv with its own dependency set.
#
# Setup:
#   python3 -m venv .venv-mcp && source .venv-mcp/bin/activate
#   pip install -r scripts/mcp/requirements.txt
#   python3 app/mcp_server.py          # stdio transport, for Claude Desktop etc.
#
mcp[cli]==2.1.1
httpx==0.28.1
